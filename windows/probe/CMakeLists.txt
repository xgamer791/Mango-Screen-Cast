cmake_minimum_required(VERSION 3.21)
project(MangoProbe CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(probe probe.cpp)
target_compile_options(probe PRIVATE /W3 /EHsc /permissive- /bigobj /utf-8)
target_compile_definitions(probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(probe PRIVATE d3d11 dxgi mfplat mfuuid ole32 dnsapi wtsapi32 windowsapp delayimp)
target_link_options(probe PRIVATE /DELAYLOAD:mfplat.dll)
